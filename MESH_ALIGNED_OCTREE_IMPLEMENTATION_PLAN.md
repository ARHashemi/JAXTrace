# Mesh-Aligned Octree Implementation Plan

**Goal**: Achieve CPU-like 100% particle retention by exploiting the Kuhn mesh's intrinsic octree structure.

**Status**: Phase 0 Complete (Epsilon fix committed for immediate testing)

## Executive Summary

### Problem Statement
Current implementation assigns elements to octree leaves by **centroid**, causing elements to span multiple leaves. Bounded radius search (±10-30 leaves) misses large elements when query points fall far from centroids. This results in 93-98% retention vs 100% for CPU octrees.

### Root Cause
1. **Single-assignment storage**: Each element stored once at centroid Morton code
2. **Morton-adjacent search**: Searches consecutive leaves in sorted array, not spatial neighbors
3. **Geometric miss**: Large elements spanning >radius leaves missed if query point ≠ centroid
4. **Float32 precision**: Tolerance 1e-10 too tight, rejects boundary particles

### Solution Strategy
Exploit the fact that the **mesh itself is generated by an octree refiner**:
- Kuhn mesh has intrinsic octree structure from adaptive refinement
- Each tetrahedron belongs to exactly ONE octree cell (cube)
- Elements have 3 axis-aligned edges (X, Y, Z) per tet
- By aligning search octree with mesh's intrinsic octree, elements never span leaves

### Expected Results
- **Retention**: 99-100% (CPU-like completeness)
- **Performance**: 25,000-30,000 p/s (faster than current 18,937 p/s)
- **Memory**: ~15-20 MB (slightly more but bounded)
- **Point-in-tet tests**: ~12 per query (vs current ~5,376) = **450× reduction**

def infer_octree_cell(elem_id, connectivity, node_positions):

def encode_morton_3d(x, y, z, max_depth=21):

                  ((y & (1 << i)) << (2 * i + 1)) | \
                  ((z & (1 << i)) << (2 * i + 2))
    return morton

# File: jaxtrace/gpu/search/mesh_octree_cell_extraction.py

"""
Mesh Octree Cell Extraction

Extract the intrinsic octree cell structure from a Kuhn-tetrahedral mesh
generated by adaptive octree refinement.

For Kuhn meshes:
- Each tetrahedron belongs to exactly one octree cell (cube)
- Cell can be inferred from element's axis-aligned bounding box
- Each cell contains 5-6 tetrahedra (Kuhn subdivision patterns)
"""

import numpy as np
from typing import Tuple
from collections import defaultdict


def extract_octree_cells(
    connectivity: np.ndarray,
    node_positions: np.ndarray,
    verbose: bool = True
) -> Tuple[np.ndarray, np.ndarray, dict]:
    """
    Extract octree cell IDs for all mesh elements.

    For each element, infers the parent octree cell (level, i, j, k) from
    the element's axis-aligned bounding box.

    Parameters
    ----------
    connectivity : ndarray, shape (n_elements, 4), int32
        Tetrahedral element connectivity

    node_positions : ndarray, shape (n_nodes, 3), float32
        Node coordinates

    verbose : bool, default=True
        Print progress information

    Returns
    -------
    cell_morton_codes : ndarray, shape (n_elements,), uint64
        Morton code of parent cell for each element

    cell_levels : ndarray, shape (n_elements,), uint8
        Octree depth level for each element's parent cell

    cell_metadata : dict
        Statistics and metadata about extracted cells:
        - 'n_unique_cells': number of unique cells
        - 'elements_per_cell': dict mapping cell_id -> element list
        - 'level_distribution': element count per depth level
        - 'cell_size_range': (min, max) cell sizes

    Notes
    -----
    - Uses tight AABB to infer cell size and level
    - Assumes cells are cubes (octree property)
    - Level = -log2(cell_size)
    - Grid indices computed at inferred level
    """
    n_elements = connectivity.shape[0]

    if verbose:
        print("="*80)
        print("Extracting Mesh Octree Cell Structure")
        print("="*80)
        print(f"  Elements: {n_elements:,}")

    # Allocate output arrays
    cell_morton_codes = np.zeros(n_elements, dtype=np.uint64)
    cell_levels = np.zeros(n_elements, dtype=np.uint8)
    cell_sizes = np.zeros(n_elements, dtype=np.float32)

    # Process each element
    for elem_id in range(n_elements):
        # Get element vertices
        node_ids = connectivity[elem_id]
        vertices = node_positions[node_ids]  # (4, 3)

        # Compute tight AABB
        bbox_min = vertices.min(axis=0)
        bbox_max = vertices.max(axis=0)
        bbox_size = bbox_max - bbox_min

        # Infer cell size (max dimension, assuming cube)
        cell_size = bbox_size.max()
        cell_sizes[elem_id] = cell_size

        # Infer level: 2^(-level) = cell_size
        if cell_size > 1e-15:
            level = int(round(-np.log2(cell_size)))
            level = np.clip(level, 0, 20)
        else:
            level = 20  # Degenerate element

        cell_levels[elem_id] = level

        # Compute grid indices at this level
        grid_scale = 2 ** level
        i = int(np.floor(bbox_min[0] * grid_scale))
        j = int(np.floor(bbox_min[1] * grid_scale))
        k = int(np.floor(bbox_min[2] * grid_scale))

        # Clamp to valid range
        max_coord = (1 << 20)  # 2^20
        i = np.clip(i, 0, max_coord - 1)
        j = np.clip(j, 0, max_coord - 1)
        k = np.clip(k, 0, max_coord - 1)

        # Encode as Morton code
        morton = encode_morton_3d(i, j, k)
        cell_morton_codes[elem_id] = morton

        # Progress reporting
        if verbose and (elem_id + 1) % 500000 == 0:
            print(f"    Processed {elem_id + 1:,}/{n_elements:,} elements...")

    if verbose:
        print(f"  ✅ Extracted cell IDs for {n_elements:,} elements")

    # Build cell-to-elements mapping
    cell_to_elements = defaultdict(list)
    for elem_id, cell_id in enumerate(cell_morton_codes):
        cell_to_elements[cell_id].append(elem_id)

    n_unique_cells = len(cell_to_elements)

    # Level distribution
    unique_levels, level_counts = np.unique(cell_levels, return_counts=True)
    level_distribution = dict(zip(unique_levels.tolist(), level_counts.tolist()))

    # Elements per cell statistics
    elements_per_cell_counts = np.array([len(elems) for elems in cell_to_elements.values()])

    # Package metadata
    cell_metadata = {
        'n_unique_cells': n_unique_cells,
        'elements_per_cell': dict(cell_to_elements),
        'level_distribution': level_distribution,
        'cell_size_range': (cell_sizes.min(), cell_sizes.max()),
        'elements_per_cell_stats': {
            'mean': float(elements_per_cell_counts.mean()),
            'median': float(np.median(elements_per_cell_counts)),
            'min': int(elements_per_cell_counts.min()),
            'max': int(elements_per_cell_counts.max()),
            'std': float(elements_per_cell_counts.std())
        }
    }

    if verbose:
        print(f"\n  Unique cells: {n_unique_cells:,}")
        print(f"  Depth range: {cell_levels.min()}-{cell_levels.max()}")
        print(f"  Elements per cell: {elements_per_cell_counts.mean():.1f} ± {elements_per_cell_counts.std():.1f}")
        print("="*80)

    return cell_morton_codes, cell_levels, cell_metadata


def encode_morton_3d(x: int, y: int, z: int) -> int:
    """
    Encode 3D grid coordinates as Morton code (Z-order curve).

    Interleaves bits of x, y, z coordinates to produce spatial hash.

    Parameters
    ----------
    x, y, z : int
        Grid coordinates (non-negative integers)

    Returns
    -------
    morton : int
        64-bit Morton code

    Notes
    -----
    - Supports up to 21 bits per coordinate (2^21 grid resolution)
    - Total: 63 bits (3 × 21)
    - Coordinates must be in range [0, 2^21)
    """
    morton = 0
    for i in range(21):  # 21 bits per coordinate
        morton |= ((x & (1 << i)) << (2 * i)) | \
                  ((y & (1 << i)) << (2 * i + 1)) | \
                  ((z & (1 << i)) << (2 * i + 2))
    return morton


def decode_morton_3d(morton: int) -> Tuple[int, int, int]:
    """
    Decode Morton code back to 3D grid coordinates.

    Parameters
    ----------
    morton : int
        64-bit Morton code

    Returns
    -------
    x, y, z : int
        Grid coordinates
    """
    x = y = z = 0
    for i in range(21):
        x |= ((morton >> (3 * i)) & 1) << i
        y |= ((morton >> (3 * i + 1)) & 1) << i
        z |= ((morton >> (3 * i + 2)) & 1) << i
    return x, y, z

"""
Cell-Based Octree Search Structure

Build and query octree structure aligned with mesh's intrinsic octree cells.
Each cell contains a small, bounded list of elements (5-6 for Kuhn meshes).

Point location algorithm:
1. Position → octree cell ID via integer quantization
2. Test all elements in that cell (typically 5-6 point-in-tet tests)
3. If not found, check spatial neighbor cells (26-connectivity)

Expected performance:
- 99-100% retention (CPU-like completeness)
- ~12 point-in-tet tests per query (vs current ~5,376)
- 450× reduction in point-in-tet tests!
"""

import numpy as np
import jax
import jax.numpy as jnp
from typing import NamedTuple
from collections import defaultdict


class CellBasedOctree(NamedTuple):
    """
    Cell-based octree search structure (GPU-compatible).

    Structure:
    - Cells sorted by Morton code for binary search
    - Each cell has pointer to element list (start, length)
    - Elements stored in contiguous array (cell-major order)

    Memory layout:
        cell_morton_codes: [cell0_morton, cell1_morton, ...] (sorted)
        cell_start: [0, 5, 11, ...]  (offset into elements_sorted)
        cell_length: [5, 6, 5, ...]  (number of elements in cell)
        elements_sorted: [elem_ids for cell0, elem_ids for cell1, ...]
    """
    cell_morton_codes: jax.Array  # (n_cells,) uint64, sorted
    cell_levels: jax.Array        # (n_cells,) uint8
    cell_start: jax.Array         # (n_cells,) int32
    cell_length: jax.Array        # (n_cells,) int32
    elements_sorted: jax.Array    # (total_entries,) int32

    # Statistics
    n_cells: int
    n_elements: int
    max_elements_per_cell: int


def build_cell_based_octree(
    cell_morton_codes: np.ndarray,
    cell_levels: np.ndarray,
    verbose: bool = True
) -> CellBasedOctree:
    """
    Build cell-based octree search structure from extracted cell IDs.

    Parameters
    ----------
    cell_morton_codes : ndarray, shape (n_elements,), uint64
        Morton code of parent cell for each element

    cell_levels : ndarray, shape (n_elements,), uint8
        Octree depth level for each element

    verbose : bool, default=True
        Print progress information

    Returns
    -------
    octree : CellBasedOctree
        GPU-compatible search structure

    Notes
    -----
    - Cells are sorted by Morton code for binary search
    - Element lists are stored contiguously for coalesced GPU access
    - Total memory: ~20 MB for 3M elements
    """
    n_elements = len(cell_morton_codes)

    if verbose:
        print("="*80)
        print("Building Cell-Based Octree Structure")
        print("="*80)
        print(f"  Elements: {n_elements:,}")

    # Group elements by cell
    if verbose:
        print("  Grouping elements by cell...")

    cell_to_elements = defaultdict(list)
    cell_to_level = {}

    for elem_id, (cell_id, level) in enumerate(zip(cell_morton_codes, cell_levels)):
        cell_to_elements[cell_id].append(elem_id)
        cell_to_level[cell_id] = level

    n_cells = len(cell_to_elements)

    if verbose:
        print(f"    Unique cells: {n_cells:,}")

    # Sort cells by Morton code
    if verbose:
        print("  Sorting cells by Morton code...")

    sorted_cells = sorted(cell_to_elements.keys())

    # Build GPU arrays
    if verbose:
        print("  Building GPU arrays...")

    cell_morton_array = np.array(sorted_cells, dtype=np.uint64)
    cell_level_array = np.array([cell_to_level[cid] for cid in sorted_cells], dtype=np.uint8)
    cell_start_array = np.zeros(n_cells, dtype=np.int32)
    cell_length_array = np.zeros(n_cells, dtype=np.int32)

    elements_list = []
    max_elements_per_cell = 0

    for i, cell_id in enumerate(sorted_cells):
        elems = cell_to_elements[cell_id]
        cell_start_array[i] = len(elements_list)
        cell_length_array[i] = len(elems)
        elements_list.extend(elems)
        max_elements_per_cell = max(max_elements_per_cell, len(elems))

    elements_sorted_array = np.array(elements_list, dtype=np.int32)

    # Upload to GPU
    if verbose:
        print("  Uploading to GPU...")

    octree = CellBasedOctree(
        cell_morton_codes=jnp.array(cell_morton_array),
        cell_levels=jnp.array(cell_level_array),
        cell_start=jnp.array(cell_start_array),
        cell_length=jnp.array(cell_length_array),
        elements_sorted=jnp.array(elements_sorted_array),
        n_cells=n_cells,
        n_elements=n_elements,
        max_elements_per_cell=max_elements_per_cell
    )

    if verbose:
        total_bytes = (
            cell_morton_array.nbytes +
            cell_level_array.nbytes +
            cell_start_array.nbytes +
            cell_length_array.nbytes +
            elements_sorted_array.nbytes

        print(f"    Cells: {n_cells:,}")
        print(f"    Max elements per cell: {max_elements_per_cell}")
        print(f"    Total element entries: {len(elements_list):,}")
        print(f"    Memory: {total_bytes:,} bytes ({total_bytes/1024**2:.1f} MB)")
        print("="*80)

    return octree


@jax.jit
def position_to_cell_morton(pos: jax.Array, level: int) -> jax.Array:
    """
    Convert 3D position to Morton code at given octree level.

    Parameters
    ----------
    pos : jax.Array, shape (3,)
        Position coordinates (in [0, 1)³ domain)

    level : int
        Octree depth level

    Returns
    -------
    morton : jax.Array, scalar uint64
        Morton code of cell containing position
    """
    # Quantize to grid at this level
    grid_scale = 2 ** level
    i = jnp.floor(pos[0] * grid_scale).astype(jnp.int32)
    j = jnp.floor(pos[1] * grid_scale).astype(jnp.int32)
    k = jnp.floor(pos[2] * grid_scale).astype(jnp.int32)

    # Clamp to valid range
    max_coord = (1 << 20)
    i = jnp.clip(i, 0, max_coord - 1)
    j = jnp.clip(j, 0, max_coord - 1)
    k = jnp.clip(k, 0, max_coord - 1)

    # Encode Morton (simplified for JIT - full version needs bit manipulation)
    # For now, use hash function (Phase 4 will implement proper Morton encoding)
    morton = i + j * 2097152 + k * 4398046511104  # Placeholder

    return morton


@jax.jit
def binary_search_cell(morton_codes: jax.Array, target: jax.Array) -> jax.Array:
    """
    Binary search for cell in sorted Morton code array.

    Returns cell index if found, -1 otherwise.
    """
    # JAX-compatible binary search
    def search_step(state, _):
        left, right, found_idx = state
        mid = (left + right) // 2
        mid_val = morton_codes[mid]

        # Update bounds
        new_left = jnp.where(mid_val < target, mid + 1, left)
        new_right = jnp.where(mid_val > target, mid, right)
        new_found = jnp.where(mid_val == target, mid, found_idx)

        return (new_left, new_right, new_found), None

    n = morton_codes.shape[0]
    init_state = (0, n, -1)

    # Run log2(n) iterations
    n_iterations = int(np.ceil(np.log2(n))) + 1
    (final_left, final_right, found_idx), _ = jax.lax.scan(
        search_step, init_state, None, length=n_iterations
    )

    # Verify found
    is_valid = (found_idx >= 0) & (found_idx < n) & (morton_codes[found_idx] == target)
    result = jnp.where(is_valid, found_idx, -1)

    return result

@jax.jit
def search_L2_cell_based(
    pos: jax.Array,
    octree: CellBasedOctree,
    connectivity: jax.Array,
    node_positions: jax.Array,
    M_inv_array: jax.Array,
    p0_array: jax.Array,
    tolerance: float = 1e-6
) -> jax.Array:
    """
    L2 search using mesh-aligned cell-based octree.

    Algorithm:
    1. Try all depth levels from finest to coarsest
    2. For each level, compute cell Morton code
    3. Binary search for cell in octree
    4. If cell exists, test all elements in that cell
    5. If found, return element ID

    Expected performance:
    - 99-100% retention (no geometric misses)
    - ~12 point-in-tet tests per query (vs 5,376 current)
    - 450× reduction in point-in-tet tests!

    Parameters
    ----------
    pos : jax.Array, shape (3,)
        Query position

    octree : CellBasedOctree
        Cell-based search structure

    connectivity : jax.Array, shape (n_elements, 4)
        Element connectivity

    node_positions : jax.Array, shape (n_nodes, 3)
        Node coordinates

    M_inv_array : jax.Array, shape (n_elements, 3, 3)
        Precomputed inverse matrices

    p0_array : jax.Array, shape (n_elements, 3)
        First vertex per element

    tolerance : float, default=1e-6
        Point-in-tet tolerance

    Returns
    -------
    elem_id : jax.Array, scalar int32
        Element ID if found, -1 otherwise
    """
    MAX_DEPTH = 20

    # Try each depth level from finest to coarsest
    for depth in range(MAX_DEPTH, -1, -1):
        # Compute cell Morton code at this depth
        grid_scale = 2 ** depth
        i = jnp.floor(pos[0] * grid_scale).astype(jnp.int32)
        j = jnp.floor(pos[1] * grid_scale).astype(jnp.int32)
        k = jnp.floor(pos[2] * grid_scale).astype(jnp.int32)

        # Encode Morton
        morton = morton_encode_jit(i, j, k)

        # Binary search for cell
        cell_idx = binary_search_cell(octree.cell_morton_codes, morton)

        # If cell exists, test all elements
        def test_cell_elements(cell_idx):
            start = octree.cell_start[cell_idx]
            length = octree.cell_length[cell_idx]

            # Test each element in cell
            def test_element(carry, offset):
                found_elem_id = carry
                elem_id = octree.elements_sorted[start + offset]

                # Point-in-tet test
                inside = point_in_tet_inverse(
                    pos, elem_id, M_inv_array, p0_array, tolerance
                )

                # Update if found
                new_found = jnp.where(inside & (found_elem_id < 0), elem_id, found_elem_id)
                return new_found, None

            found_elem_id, _ = jax.lax.scan(test_element, -1, jnp.arange(length))
            return found_elem_id

        # Test cell if it exists
        elem_id = jax.lax.cond(
            cell_idx >= 0,
            lambda: test_cell_elements(cell_idx),
            lambda: -1
        )

        # Return if found
        if elem_id >= 0:
            return elem_id

    # Not found in any cell
    return -1

    """
    Encode 3D grid coordinates as Morton code (JIT-compatible).

    Uses bit interleaving for spatial hashing.
    """
    # Simplified Morton encoding for JIT
    # Full implementation uses bit manipulation

    # For now, use magic number method (fast approximation)
    # TODO: Implement full bit-interleaving in Phase 4 refinement

    morton = i.astype(jnp.uint64)
    morton = morton | (j.astype(jnp.uint64) << 21)
